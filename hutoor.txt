
#!/usr/bin/env bash
set -euo pipefail

help() {
    cat << EOF
Hűtőőr - Kollégiumi Hűtő Nyomozó Eszköz

Használat:
  ./hutoor.sh -f FÁJL [kapcsolók]
  ./hutoor.sh FÁJL [kapcsolók]

Kapcsolók:
  -f, --file FÁJL       Logfájl megadása
  -h, --help            Súgó
  --attekintes          Áttekintés
  --kivel NÉV           Üzenetek adott résztvevőtől vagy neki
  --beszelgetes A B     Két résztvevő privát beszélgetése
  --export MAPPA        Exportálás résztvevőnként
  --keres MINTA         Keresés (regex)
  --tol DÁTUM           Kezdődátum
  --ig DÁTUM            Záródátum
  --csatorna NÉV        Csatornaszűrés
  --csak-privat         Csak privát üzenetek
  --gyanusitott         Gyanúsítottak rangsorolása
EOF
    exit 0
}

LOGFILE=""
COMMAND="attekintes"
ARG1=""
ARG2=""
KERES_MINTA=""
TOL_DATUM=""
IG_DATUM=""
CSATORNA_SZURES=""
CSAK_PRIVAT=0
ARGS=()

error() {
    echo "Hiba: $1" >&2
    exit 1
}

while [[ $# -gt 0 ]]; do
    case "$1" in
        --help|-h)
            help
            ;;
        -f|--file)
            [[ $# -ge 2 ]] || error "Hiányzik a fájl neve!"
            LOGFILE="$2"
            shift 2
            ;;
        --attekintes)
            COMMAND="attekintes"
            shift
            ;;
        --kivel)
            [[ $# -ge 2 ]] || error "Hiányzik a résztvevő neve!"
            COMMAND="kivel"
            ARG1="$2"
            shift 2
            ;;
        --beszelgetes)
            [[ $# -ge 3 ]] || error "Két résztvevő neve szükséges!"
            COMMAND="beszelgetes"
            ARG1="$2"
            ARG2="$3"
            shift 3
            ;;
        --export)
            [[ $# -ge 2 ]] || error "Hiányzik az exportmappa!"
            COMMAND="export"
            ARG1="$2"
            shift 2
            ;;
        --keres)
            [[ $# -ge 2 ]] || error "Hiányzik a keresési minta!"
            KERES_MINTA="$2"
            shift 2
            ;;
        --tol)
            [[ $# -ge 2 ]] || error "Hiányzik a kezdődátum!"
            TOL_DATUM="$2"
            shift 2
            ;;
        --ig)
            [[ $# -ge 2 ]] || error "Hiányzik a záródátum!"
            IG_DATUM="$2"
            shift 2
            ;;
        --csatorna)
            [[ $# -ge 2 ]] || error "Hiányzik a csatorna neve!"
            CSATORNA_SZURES="$2"
            shift 2
            ;;
        --csak-privat)
            CSAK_PRIVAT=1
            shift
            ;;
        --gyanusitott)
            COMMAND="gyanusitott"
            shift
            ;;
        -*)
            echo "Hiba: Ismeretlen kapcsoló: $1" >&2
            echo "Használd a --help kapcsolót a súgóért." >&2
            exit 1
            ;;
        *)
            ARGS+=("$1")
            shift
            ;;
    esac
done

if [[ -z "$LOGFILE" && ${#ARGS[@]} -gt 0 ]]; then
    LOGFILE="${ARGS[0]}"
fi

[[ -n "$LOGFILE" ]] || error "Nincs megadva logfájl!"
[[ -r "$LOGFILE" && ! -d "$LOGFILE" ]] || error "A(z) '$LOGFILE' fájl nem létezik vagy nem olvasható."

validate_date() {
    local d="$1"

    [[ -z "$d" ]] && return 0

    if [[ ! "$d" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}([T][0-9]{2}:[0-9]{2}:[0-9]{2})?$ ]]; then
        echo "Figyelmeztetés: Érvénytelen dátumformátum: $d" >&2
        return 1
    fi

    local date_part="${d%%T*}"

    if ! date -d "$date_part" '+%Y-%m-%d' >/dev/null 2>&1 ||
       [[ "$(date -d "$date_part" '+%Y-%m-%d')" != "$date_part" ]]; then
        echo "Figyelmeztetés: Érvénytelen dátum: $d" >&2
        return 1
    fi

    if [[ "$d" == *T* ]]; then
        if ! date -d "$d" '+%Y-%m-%dT%H:%M:%S' >/dev/null 2>&1 ||
           [[ "$(date -d "$d" '+%Y-%m-%dT%H:%M:%S')" != "$d" ]]; then
            echo "Figyelmeztetés: Érvénytelen dátum vagy idő: $d" >&2
            return 1
        fi
    fi

    return 0
}

validate_date "$TOL_DATUM" || TOL_DATUM=""
validate_date "$IG_DATUM" || IG_DATUM=""

if [[ -n "$KERES_MINTA" ]]; then
    if ! awk -v pattern="$KERES_MINTA" 'BEGIN { result = ("" ~ pattern) }' </dev/null 2>/dev/null; then
        error "Érvénytelen keresési minta!"
    fi
fi

case "$COMMAND" in
    kivel)
        [[ -n "$ARG1" ]] || error "Add meg a résztvevő nevét!"
        ;;
    beszelgetes)
        [[ -n "$ARG1" && -n "$ARG2" ]] || error "Két résztvevő szükséges!"
        ;;
    export)
        [[ -n "$ARG1" ]] || error "Add meg az exportmappát!"
        mkdir -p -- "$ARG1" || error "Nem sikerült létrehozni az exportmappát."
        ;;
esac

awk \
    -v cmd="$COMMAND" \
    -v arg1="$ARG1" \
    -v arg2="$ARG2" \
    -v keres="$KERES_MINTA" \
    -v tol="$TOL_DATUM" \
    -v ig="$IG_DATUM" \
    -v csatorna_szures="$CSATORNA_SZURES" \
    -v csak_privat="$CSAK_PRIVAT" \
    -v export_mappa="$ARG1" '

function add_person(name) {
    if (!(name in person_exists)) {
        person_exists[name] = 1
        people[++person_count] = name
    }
}

function add_channel(name) {
    if (!(name in channel_exists)) {
        channel_exists[name] = 1
        channels[++channel_count] = name
    }
}

function matches_filter(i) {
    if (tol != "") {
        if (length(tol) == 10) {
            if (substr(timestamp[i], 1, 10) < tol)
                return 0
        } else if (timestamp[i] < tol)
            return 0
    }

    if (ig != "") {
        if (length(ig) == 10) {
            if (substr(timestamp[i], 1, 10) > ig)
                return 0
        } else if (timestamp[i] > ig)
            return 0
    }

    if (csatorna_szures != "" && channel[i] != csatorna_szures)
        return 0

    if (csak_privat && !private[i])
        return 0

    if (keres != "" && tolower(message[i]) !~ tolower(keres))
        return 0

    if (cmd == "kivel" && sender[i] != arg1 && receiver[i] != arg1)
        return 0

    if (cmd == "beszelgetes") {
        if (!private[i])
            return 0

        if (!((sender[i] == arg1 && receiver[i] == arg2) ||
              (sender[i] == arg2 && receiver[i] == arg1)))
            return 0
    }

    return 1
}

{
    timestamp_value = substr($0, 1, 19)
    rest = substr($0, 21)

    if (timestamp_value !~ /^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}$/)
        next

    is_private = 0
    from = ""
    to = ""
    ch = ""
    msg = ""

    if (match(rest, /^#[^[:space:]]+[[:space:]]+<[^>]+>/)) {
        ch = substr(rest, 2, index(rest, " ") - 2)

        temp = substr(rest, index(rest, "<") + 1)
        from = substr(temp, 1, index(temp, ">") - 1)
        msg = substr(temp, index(temp, ">") + 1)

        sub(/^[[:space:]]+/, "", msg)
    }
    else if (match(rest, /^@[^[:space:]]+[[:space:]]+->[[:space:]]+@[^[:space:]]+[[:space:]]*:/)) {
        is_private = 1

        temp = substr(rest, 2)
        from = substr(temp, 1, index(temp, " ") - 1)

        temp = substr(rest, index(rest, "->") + 2)
        sub(/^[[:space:]]+@/, "", temp)

        to = substr(temp, 1, index(temp, " ") - 1)
        sub(/^:[[:space:]]*/, "", temp)
        msg = substr(temp, index(temp, ":") + 1)
        sub(/^[[:space:]]+/, "", msg)
    }
    else
        next

    total++

    timestamp[total] = timestamp_value
    sender[total] = from
    receiver[total] = to
    channel[total] = ch
    message[total] = msg
    private[total] = is_private

    add_person(from)

    if (is_private) {
        add_person(to)
        private_total++
        private_sent[from]++
    } else {
        add_channel(ch)
    }

    message_count[from]++

    if (msg ~ /(eltunt|megette|maradekom|sajtom|lop)/)
        complaints[from]++

    hour = substr(timestamp_value, 12, 2)
    if (hour >= 1 && hour < 4)
        night[from]++

    if (tolower(msg) ~ /(en biztos nem|nem voltam|egesz nap|csak epp lementem)/)
        excuses[from]++

    if (first == "" || timestamp_value < first)
        first = timestamp_value

    if (last == "" || timestamp_value > last)
        last = timestamp_value
}

END {
    if (total == 0) {
        if (cmd == "kivel")
            printf "Nincs ilyen résztvevő: %s\n", arg1
        else if (cmd == "beszelgetes")
            print "Nincs ilyen résztvevő a megadott nevek között."
        else
            print "A fájl üres, vagy nem tartalmaz feldolgozható üzeneteket."

        exit
    }

    for (i = 1; i <= total; i++) {
        if (matches_filter(i))
            selected[++selected_count] = i
    }

    if (cmd == "export") {
        for (p = 1; p <= person_count; p++) {
            name = people[p]
            filename = export_mappa "/" name ".txt"

            printf "" > filename
            close(filename)

            for (i = 1; i <= total; i++) {
                if (sender[i] == name || receiver[i] == name) {
                    if (private[i])
                        line = timestamp[i] " @" sender[i] " -> @" receiver[i] " : " message[i]
                    else
                        line = timestamp[i] " #" channel[i] " <" sender[i] "> " message[i]

                    print line >> filename
                }
            }

            close(filename)
        }

        printf "Exportálás kész: %d résztvevő fájljai a(z) %s mappában.\n", person_count, export_mappa
        exit
    }

    if (cmd == "attekintes" &&
        keres == "" && tol == "" && ig == "" &&
        csatorna_szures == "" && !csak_privat) {

        printf "HŰTŐŐR - ÁTTEKINTÉS\n\n"
        printf "Összes üzenet:       %d\n", total
        printf "Privát üzenetek:     %d\n", private_total
        printf "Résztvevők száma:    %d\n", person_count
        printf "Első üzenet:         %s\n", first
        printf "Utolsó üzenet:       %s\n", last
        printf "Szűrt találatok:     %d\n\n", selected_count

        print "Résztvevők toplistája:"
        print "------------------------------"

        for (rank = 1; rank <= person_count; rank++) {
            best = ""
            best_count = -1

            for (p = 1; p <= person_count; p++) {
                name = people[p]

                if (!(used[name]) && message_count[name] > best_count) {
                    best = name
                    best_count = message_count[name]
                }
            }

            used[best] = 1
            printf "%-20s %5d üzenet\n", best, best_count
        }

        print "\nCsatornánkénti bontás:"
        print "------------------------------"

        for (p = 1; p <= channel_count; p++) {
            name = channels[p]
            count = 0

            for (i = 1; i <= total; i++) {
                if (!private[i] && channel[i] == name)
                    count++
            }

            printf "%-20s %5d üzenet\n", name, count
        }
    }
    else if (cmd == "gyanusitott") {
        for (p = 1; p <= person_count; p++) {
            name = people[p]

            complaint_points = complaints[name] == 0 ? 5 : (complaints[name] == 1 ? 3 : 0)
            night_points = night[name] > 5 ? 5 : night[name]
            excuse_points = excuses[name] * 2
            if (excuse_points > 10)
                excuse_points = 10

            private_points = private_sent[name] > 5 ? 5 : private_sent[name]

            scores[name] = complaint_points + night_points + excuse_points + private_points
        }

        print "GYANÚSÍTOTTAK TOP 3"
        print "----------------------------------------"

        for (rank = 1; rank <= 3 && rank <= person_count; rank++) {
            best = ""
            best_score = -1

            for (p = 1; p <= person_count; p++) {
                name = people[p]

                if (!(suspect_used[name]) && scores[name] > best_score) {
                    best = name
                    best_score = scores[name]
                }
            }

            suspect_used[best] = 1

            printf "\n%d. %s - %d pont\n", rank, best, best_score
            printf "   Kevés panaszkodás: %d panasz\n", complaints[best]
            printf "   Hajnali aktivitás: %d üzenet\n", night[best]
            printf "   Kéretlen mentegetőzés: %d alkalom\n", excuses[best]
            printf "   Privát üzenetek: %d\n", private_sent[best]
        }

        print "\nPontozás:"
        print "Kevés panasz: 0 panasz = 5 pont, 1 panasz = 3 pont."
        print "Hajnali aktivitás: üzenetenként 1 pont, maximum 5."
        print "Mentegetőzés: alkalmanként 2 pont, maximum 10."
        print "Privát üzenetek: üzenetenként 1 pont, maximum 5."
    }
    else {
        if (cmd == "kivel" && !(arg1 in person_exists)) {
            printf "Nincs ilyen résztvevő: %s\n", arg1
            exit
        }

        if (cmd == "beszelgetes" &&
            (!(arg1 in person_exists) || !(arg2 in person_exists))) {
            print "Nincs ilyen résztvevő a megadott nevek között."
            exit
        }

        for (j = 1; j <= selected_count; j++) {
            i = selected[j]

            if (private[i])
                printf "%s @%s -> @%s : %s\n", timestamp[i], sender[i], receiver[i], message[i]
            else
                printf "%s #%s <%s> %s\n", timestamp[i], channel[i], sender[i], message[i]
        }

        printf "\nTalálatok száma: %d\n", selected_count
    }
}
' "$LOGFILE"
