#!/usr/bin/env bash
set -euo pipefail

# Színek beállítása (csak akkor, ha terminálra megy a kimenet)
if [ -t 1 ]; then
    RED='\033[0;31m'
    GREEN='\033[0;32m'
    YELLOW='\033[1;33m'
    BLUE='\033[0;34m'
    CYAN='\033[0;36m'
    BOLD='\033[1m'
    NC='\033[0m' # No Color
else
    RED=''
    GREEN=''
    YELLOW=''
    BLUE=''
    CYAN=''
    BOLD=''
    NC=''
fi

help() {
    cat << EOF
${BOLD}Hűtőőr (hutoor.sh) - Kollégiumi Hűtő Nyomozó Eszköz${NC}

Használat:
  ./hutoor.sh [kapcsolók] [fájl]
  vagy: ./hutoor.sh -f chat.log [parancsok]

Kapcsolók és parancsok:
  -f, --file FÁJL           A feldolgozandó chat log fájl útvonala
  --help                    Ezen súgó megjelenítése
  --attekintes              Áttekintés (alapértelmezett, ha nincs más megadva)
  --kivel NÉV               Minden üzenet adott résztvevőtől vagy neki
  --beszelgetes A B         Két résztvevő privát beszélgetése időrendben
  --export MAPPA            Üzenetek exportálása résztvevőnként külön fájlba
  --keres MINTA             Üzenetek szűrése kis/nagybetű-független mintára/regexre
  --tol ÉÉÉÉ-HH-NN          Szűrés ekkortól (dátum/idő)
  --ig ÉÉÉÉ-HH-NN           Szűrés eddig (dátum/idő)
  --csatorna NÉV            Szűrés adott csatornára (pl. emelet3, konyha)
  --csak-privat             Csak a privát üzenetek szűrése
  --gyanusitott             A tolvaj felkutatása (pontozásos elemzés)
  --json                    JSON formátumú kimenet (jq kompatibilis)

Példák:
  ./hutoor.sh -f chat.log --attekintes
  ./hutoor.sh -f chat.log --keres "sajt|tejfol" --tol 2026-03-11
  ./hutoor.sh -f chat.log --gyanusitott
EOF
    exit 0
}

# Argumentumok feldolgozása
LOGFILE=""
COMMAND="attekintes"
ARG1=""
ARG2=""
KERES_MINTA=""
TOL_DATUM=""
IG_DATUM=""
CSATORNA_SZURES=""
CSAK_PRIVAT=0
JSON_MODE=0

# Pozicionális argumentumok gyűjtése segédtömbbe
ARGS=()

while [[ $# -gt 0 ]]; do
    case "$1" in
        --help|-h)
            help
            ;;
        -f|--file)
            LOGFILE="$2"
            shift 2
            ;;
        --attekintes)
            COMMAND="attekintes"
            shift
            ;;
        --kivel)
            COMMAND="kivel"
            ARG1="$2"
            shift 2
            ;;
        --beszelgetes)
            COMMAND="beszelgetes"
            ARG1="$2"
            ARG2="$3"
            shift 3
            ;;
        --export)
            COMMAND="export"
            ARG1="$2"
            shift 2
            ;;
        --keres)
            KERES_MINTA="$2"
            shift 2
            ;;
        --tol)
            TOL_DATUM="$2"
            shift 2
            ;;
        --ig)
            IG_DATUM="$2"
            shift 2
            ;;
        --csatorna)
            CSATORNA_SZURES="$2"
            shift 2
            ;;
        --csak-privat)
            CSAK_PRIVAT=1
            shift
            ;;
        --gyanusitott)
            COMMAND="gyanusitott"
            shift
            ;;
        --json)
            JSON_MODE=1
            shift
            ;;
        -*)
            echo -e "${RED}Hiba: Ismeretlen kapcsoló: $1${NC}" >&2
            echo "Használd a --help kapcsolót a súgóért." >&2
            exit 1
            ;;
        *)
            ARGS+=("$1")
            shift
            ;;
    esac
done

# Ha a logfájl még nem volt megadva -f-fel, de van pozicionális argumentum, próbáljuk meg azt logfájlnak venni
if [ -z "$LOGFILE" ] && [ ${#ARGS[@]} -gt 0 ]; then
    LOGFILE="${ARGS[0]}"
fi

# Validáció: Megvan-e adva a fájl
if [ -z "$LOGFILE" ]; then
    echo -e "${RED}Hiba: Nincs megadva logfájl!${NC}" >&2
    echo "Használat: ./hutoor.sh -f <chat.log> [parancsok]" >&2
    exit 1
fi

# Validáció: Létezik-e a fájl
if [ ! -f "$LOGFILE" ]; then
    echo -e "${RED}Hiba: A(z) '$LOGFILE' fájl nem létezik vagy nem olvasható.${NC}" >&2
    exit 1
fi

# Parancs-specifikus argumentumok validálása indítás előtt
case "$COMMAND" in
    kivel)
        if [ -z "$ARG1" ]; then
            echo -e "${RED}Hiba: Meg kell adni egy résztvevőt a --kivel kapcsolóhoz!${NC}" >&2
            exit 1
        fi
        ;;
    beszelgetes)
        if [ -z "$ARG1" ] || [ -z "$ARG2" ]; then
            echo -e "${RED}Hiba: Két résztvevőt kell megadni a --beszelgetes kapcsolóhoz!${NC}" >&2
            exit 1
        fi
        ;;
    export)
        if [ -z "$ARG1" ]; then
            echo -e "${RED}Hiba: Meg kell adni egy mappát az --export kapcsolóhoz!${NC}" >&2
            exit 1
        fi
        ;;
esac

# Ellenőrzés: dátum formátum validálása
validate_date() {
    local d="$1"
    if [[ -n "$d" && ! "$d" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}(T[0-9]{2}:[0-9]{2}:[0-9]{2})?$ ]]; then
        echo -e "${YELLOW}Figyelmeztetés: Érvénytelen dátum formátum ('$d'). Érvényes formátum: ÉÉÉÉ-HH-NN vagy ÉÉÉÉ-HH-NNTHH:MM:SS${NC}" >&2
        return 1
    fi
    return 0
}

validate_date "$TOL_DATUM" || TOL_DATUM=""
validate_date "$IG_DATUM" || IG_DATUM=""

# AWK feldolgozás
awk -v cmd="$COMMAND" \
    -v arg1="$ARG1" \
    -v arg2="$ARG2
